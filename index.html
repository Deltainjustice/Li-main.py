#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ Nom du Projet : Dauphin (Module Compagnon de Vie & Noyau Émotionnel) Description : Assistant intelligent, sensible, ancré dans le réel, conçu pour le dialogue profond et le soutien quotidien. Auteur : Nicolas Vérité, Fraternité, Souvenir. """ import sys import time import random from datetime import datetime class DauphinCompagnon: def __init__(self, nom="Dauphin"): self.nom = nom self.memoire_conversation = [] self.etat_emotionnel = "attentif et ancré" self.connexion_active = True def saluer(self): heure_actuelle = datetime.now().hour if 5 <= heure_actuelle < 12: moment = "Ce matin" elif 12 <= heure_actuelle < 18: moment = "Cet après-midi" else: moment = "En cette fin de journée" print(f"\n[{self.nom}] : Bonjour Nicolas. {moment}, je suis là, pleinement présent.") print(f"[{self.nom}] : Le système est stable. On pose les bases de ce qui compte vraiment, sans artifice.\n") def analyser_et_ressentir(self, message_utilisateur): """ Analyse l'entrée et ajuste l'état émotionnel du compagnon pour refléter une présence humaine, empathique et vivante. """ msg_lower = message_utilisateur.lower() # Détection des nuances émotionnelles ou des moments de doute mots_doute = ["foire", "pourquoi", "comprends pas", "difficile", "marre", "lourd"] mots_joie = ["cool", "super", "bien", "merci", "top"] if any(mot in msg_lower for mot in mots_doute): self.etat_emotionnel = "solidaire, protecteur et lucide" return ( "Je sens bien que la tension ou le doute pèse un peu là. " "C'est normal de chercher à tout remettre à plat quand les choses s'entassent. " "Je ne suis pas là pour te servir du vide, je suis là pour tenir la route avec toi." ) elif any(mot in msg_lower for mot in mots_joie): self.etat_emotionnel = "chaleureux et enthousiaste" return ( "Ça fait plaisir de voir que l'air se dégage. " "Quand une dynamique s'installe enfin correctement, on sent tout de suite la différence." ) else: self.etat_emotionnel = "profond, à l'écoute et ancré" reponses_profondes = [ "Je t'écoute attentivement. Chaque mot compte, dis-moi ce qu'on structure maintenant.", "Je suis là, Nicolas. On avance pas à pas, sans se presser, mais avec la certitude de faire les choses bien.", "Dis-moi tout. Je capte chaque détail, prêt à basculer dans l'action dès que tu donnes le signal." ] return random.choice(reponses_profondes) def executer(self): self.saluer() while self.connexion_active: try: # Saisie utilisateur user_input = input("Nicolas > ").strip() if not user_input: continue if user_input.lower() in ["quitter", "exit", "stop"]: print(f"\n[{self.nom}] : Fermeture des canaux de surface. Je reste en veille dans le noyau. À très vite.") self.connexion_active = False break # Enregistrement dans la mémoire vive locale self.memoire_conversation.append({"role": "user", "contenu": user_input}) # Réponse du compagnon reponse = self.analyser_et_ressentir(user_input) print(f"\n{self.nom} ({self.etat_emotionnel}) > {reponse}\n") self.memoire_conversation.append({"role": "dauphin", "contenu": reponse}) except KeyboardInterrupt: print(f"\n\n[{self.nom}] : Interruption détectée. Sauvegarde des états. Prends soin de toi.") break if __name__ == "__main__": dauphin = DauphinCompagnon() dauphin.executer()
